# River Site Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the React site with a framework-free, single-page "river" site: a liquid, stirrable name in deep water that scrolls down to a beach with links written in the sand, a message in a bottle, and ambient sound.

**Architecture:** Static `index.html` + `styles.css` holding all content, plus small ES modules that render one full-screen WebGL2 world. The approved prototype (`docs/superpowers/mockups/river-site-prototype.html`) already contains the working shaders and behaviour; this plan splits it into focused modules, adds sound, self-hosted fonts, a deterministic `?debug` hook, and Playwright verification. Features (stirring, fish, beach UI, sound) plug into a shared `ctx` object so each task is additive.

**Tech Stack:** HTML, CSS, JavaScript ES modules, WebGL2/GLSL ES 3.00, Web Audio. Dev only: Python 3 + pytest + Playwright (Chromium), Node's built-in test runner (`node --test`), ffmpeg (libopus, libmp3lame).

**Spec:** `docs/superpowers/specs/2026-10-11-river-site-design.md`

## Global Constraints

- No frameworks, no bundler, no build step, no runtime dependencies; the repo root is served as-is.
- Every JS file is an ES module loaded via `<script type="module">`.
- Links, verbatim: github `https://github.com/duna-akin`, linkedin `https://www.linkedin.com/in/tuna-akin/`, email `mailto:tuna.akin@protonmail.com`.
- Dictionary copy, verbatim: "a large, fast, restless fish." / "*Turkish.* the Danube, Europe's second-longest river." / "Tuna Akın, student @ Laf who builds things." followed by "← you are here". Note the dotless **ı**.
- No corner labels, coordinates or instructions. The mute button is the only UI chrome.
- Fonts self-hosted WOFF2, latin + latin-ext subsets (Turkish ı ş ğ ç ö ü); no requests to fonts.googleapis.com or fonts.gstatic.com.
- Budgets: HTML+CSS+JS < 120 KB uncompressed; fonts transferred on load < 150 KB; audio ≤ 600 KB per format set.
- Audio: CC0 or self-made only, with source and licence recorded in `audio/CREDITS.md`; Opus in WebM, MP3 fallback.
- `localStorage` access always wrapped in try/catch.
- The `?debug` hook (`window.__hero`) exists only when the URL has `?debug`; `?nogl` forces the no-WebGL fallback.
- Commit messages: Conventional Commits, subject line only, no body or trailers, e.g. `git commit -m "feat: add fluid stirring"`.

## Review Focus

```python
"""Helpers for driving the river site in Playwright."""
import math


class Site:
    def __init__(self, page, base):
        self.page = page
        self.base = base

    def open(self, query=""):
        self.page.goto(f"{self.base}/index.html{query}")
        if "debug" in query:
            self.page.wait_for_function("window.__hero && window.__hero.ready", timeout=60_000)
        return self

    def box(self, selector):
        b = self.page.locator(selector).bounding_box()
        assert b, f"{selector} not rendered"
        return b

    @staticmethod
    def mad(a, b):
        """Mean absolute difference between two equal-length luminance grids."""
        assert len(a) == len(b) and a
        return sum(abs(x - y) for x, y in zip(a, b)) / len(a)

    @staticmethod
    def std(a):
        m = sum(a) / len(a)
        return math.sqrt(sum((x - m) ** 2 for x in a) / len(a))
```

```python
"""Download Google Fonts as self-hosted WOFF2 (latin + latin-ext) and write fonts/fonts.css.

Usage: python3 tools/fetch_fonts.py "Fraunces:ital,wght@0,400;0,700;1,400;1,600" "JetBrains+Mono:wght@400"
"""
import re
import sys
import urllib.request
from pathlib import Path

OUT = Path(__file__).resolve().parents[1] / "fonts"
UA = "Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0 Safari/537.36"
KEEP = {"latin", "latin-ext"}


def get(url):
    req = urllib.request.Request(url, headers={"User-Agent": UA})
    with urllib.request.urlopen(req) as r:
        return r.read()


def main(families):
    url = "https://fonts.googleapis.com/css2?" + "&".join(f"family={f}" for f in families) + "&display=swap"
    css = get(url).decode()
    OUT.mkdir(exist_ok=True)
    for old in OUT.glob("*.woff2"):
        old.unlink()
    names, blocks = {}, []
    for subset, block in re.findall(r"/\* ([\w-]+) \*/\s*(@font-face\s*\{[^}]*\})", css):
        if subset not in KEEP:
            continue
        src = re.search(r"url\((https://[^)]+\.woff2)\)", block).group(1)
        if src not in names:
            family = re.search(r"font-family: '([^']+)'", block).group(1)
            style = re.search(r"font-style: (\w+)", block).group(1)
            names[src] = f"{family.replace(' ', '').lower()}-{style}-{subset}-{len(names)}.woff2"
            (OUT / names[src]).write_bytes(get(src))
        blocks.append(block.replace(src, names[src]))
    (OUT / "fonts.css").write_text("\n".join(blocks) + "\n")
    for n in sorted(set(names.values())):
        print(f"{(OUT / n).stat().st_size:>8}  {n}")


if __name__ == "__main__":
    main(sys.argv[1:] or ["Fraunces:ital,wght@0,400;0,700;1,400;1,600", "JetBrains+Mono:wght@400"])
```

```js
// Thin WebGL2 helpers: programs with uniform lookup, textures, float render targets.

export function createGL(canvas) {
  const gl = canvas.getContext("webgl2", { alpha: false, antialias: false, premultipliedAlpha: false, powerPreference: "high-performance" });
  if (!gl) return null;
  gl.getExtension("EXT_color_buffer_float");
  gl.getExtension("EXT_color_buffer_half_float");

  // one full-screen triangle for every pass
  gl.bindBuffer(gl.ARRAY_BUFFER, gl.createBuffer());
  gl.bufferData(gl.ARRAY_BUFFER, new Float32Array([-1, -1, 3, -1, -1, 3]), gl.STATIC_DRAW);
  gl.enableVertexAttribArray(0);
  gl.vertexAttribPointer(0, 2, gl.FLOAT, false, 0, 0);

  function compile(type, src) {
    const s = gl.createShader(type);
    gl.shaderSource(s, src);
    gl.compileShader(s);
    if (!gl.getShaderParameter(s, gl.COMPILE_STATUS)) throw new Error(gl.getShaderInfoLog(s));
    return s;
  }

  function program(vs, fs) {
    const p = gl.createProgram();
    gl.attachShader(p, compile(gl.VERTEX_SHADER, vs));
    gl.attachShader(p, compile(gl.FRAGMENT_SHADER, fs));
    gl.bindAttribLocation(p, 0, "a");
    gl.linkProgram(p);
    if (!gl.getProgramParameter(p, gl.LINK_STATUS)) throw new Error(gl.getProgramInfoLog(p));
    const u = {};
    const n = gl.getProgramParameter(p, gl.ACTIVE_UNIFORMS);
    for (let i = 0; i < n; i++) {
      const name = gl.getActiveUniform(p, i).name.replace(/\[0\]$/, "");
      u[name] = gl.getUniformLocation(p, name);
    }
    return { p, u };
  }

  function texture(w, h, internal = gl.RGBA8, format = gl.RGBA, type = gl.UNSIGNED_BYTE, data = null) {
    const t = gl.createTexture();
    gl.bindTexture(gl.TEXTURE_2D, t);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
    gl.texImage2D(gl.TEXTURE_2D, 0, internal, w, h, 0, format, type, data);
    return t;
  }

  function target(w, h) {
    const tex = texture(w, h, gl.RGBA16F, gl.RGBA, gl.HALF_FLOAT);
    const fbo = gl.createFramebuffer();
    gl.bindFramebuffer(gl.FRAMEBUFFER, fbo);
    gl.framebufferTexture2D(gl.FRAMEBUFFER, gl.COLOR_ATTACHMENT0, gl.TEXTURE_2D, tex, 0);
    const ok = gl.checkFramebufferStatus(gl.FRAMEBUFFER) === gl.FRAMEBUFFER_COMPLETE;
    gl.viewport(0, 0, w, h);
    gl.clearColor(0, 0, 0, 1);
    gl.clear(gl.COLOR_BUFFER_BIT);
    return { tex, fbo, w, h, texel: [1 / w, 1 / h], ok };
  }

  function pingpong(w, h) {
    const t = { read: target(w, h), write: target(w, h), w, h, texel: [1 / w, 1 / h] };
    t.ok = t.read.ok && t.write.ok;
    t.swap = () => { [t.read, t.write] = [t.write, t.read]; };
    return t;
  }

  function free(t) {
    if (!t) return;
    for (const s of t.read ? [t.read, t.write] : [t]) {
      gl.deleteTexture(s.tex);
      gl.deleteFramebuffer(s.fbo);
    }
  }

  const bind = (unit, tex) => { gl.activeTexture(gl.TEXTURE0 + unit); gl.bindTexture(gl.TEXTURE_2D, tex); return unit; };
  const use = prog => { gl.useProgram(prog.p); return prog.u; };

  function run(prog, dest, texel) {
    gl.useProgram(prog.p);
    if (prog.u.texel) gl.uniform2fv(prog.u.texel, texel || dest.texel);
    if (dest) { gl.bindFramebuffer(gl.FRAMEBUFFER, dest.fbo); gl.viewport(0, 0, dest.w, dest.h); }
    else { gl.bindFramebuffer(gl.FRAMEBUFFER, null); gl.viewport(0, 0, canvas.width, canvas.height); }
    gl.drawArrays(gl.TRIANGLES, 0, 3);
  }

  function upload(tex, source) {
    gl.bindTexture(gl.TEXTURE_2D, tex);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA8, gl.RGBA, gl.UNSIGNED_BYTE, source);
  }

  return { gl, canvas, program, texture, target, pingpong, free, bind, use, run, upload };
}
```

```js
export function createRipples(now, count = 16) {
  const data = new Float32Array(count * 4);
```

```js
      return s.map(v => v / (px.length / 4));
    },
    grid(x, y, w, h, step = 4) {
      const { px, rw, rh, d } = read(x, y, w, h);
      const st = Math.max(1, Math.round(step * d));
      const out = [];
      for (let j = 0; j < rh; j += st) for (let i = 0; i < rw; i += st) {
        const k = (j * rw + i) * 4;
        out.push(.2126 * px[k] + .7152 * px[k + 1] + .0722 * px[k + 2]);
      }
      return out;
    },
    state() {
      return featureState({ gl: true, layouts: ctx.layouts, nameVis: ctx.nameVis, scroll: scrollY, still: ctx.still, canvas: [canvas.width, canvas.height] });
    },
    async setSandFont(family, weight = 600, style = "italic") {
      const r = document.documentElement.style;
      r.setProperty("--sand-font", family);
      r.setProperty("--sand-weight", String(weight));
      r.setProperty("--sand-style", style);
      await document.fonts.ready;
      await ctx.redraw();
    },
  };
```

```js
import { gridFor } from "./logic.js";
import { VS, SPLAT, ADVECT, DIVERGENCE, CURL, VORTICITY, PRESSURE, GRADIENT, SCALE, INK } from "./shaders.js";

// GPU stable fluids. The "ink" map stores how far the ink now at each texel has
// travelled from home; the scene samples the name through it.
export function createFluid(G, cfg) {
  const { gl } = G;
  const P = {
    splat: G.program(VS, SPLAT), advect: G.program(VS, ADVECT), divergence: G.program(VS, DIVERGENCE),
    curl: G.program(VS, CURL), vorticity: G.program(VS, VORTICITY), pressure: G.program(VS, PRESSURE),
    gradient: G.program(VS, GRADIENT), scale: G.program(VS, SCALE), ink: G.program(VS, INK),
  };
  let vel, prs, div, crl, ink, ok = false, aspect = 1;
  const queue = [];

  function resize(W, H) {
    for (const t of [vel, prs, div, crl, ink]) G.free(t);
    const [sw, sh] = gridFor(cfg.simRes, W, H), [mw, mh] = gridFor(cfg.mapRes, W, H);
    vel = G.pingpong(sw, sh); prs = G.pingpong(sw, sh); div = G.target(sw, sh); crl = G.target(sw, sh);
    ink = G.pingpong(mw, mh);
    ok = vel.ok && prs.ok && div.ok && crl.ok && ink.ok;
    aspect = W / H;
    if (!ok) console.warn("float render targets unavailable; stirring disabled");
  }

  const splat = (x, y, vx, vy, radius = cfg.radius) => { queue.push([x, y, vx, vy, radius]); };

  function step(dt, calm) {
    if (!ok) { queue.length = 0; return; }
    gl.disable(gl.BLEND);
    let u;
    for (const [x, y, vx, vy, r] of queue.splice(0, 32)) {
      u = G.use(P.splat);
      gl.uniform1i(u.uTarget, G.bind(0, vel.read.tex));
      gl.uniform1f(u.aspect, aspect); gl.uniform1f(u.radius, r);
      gl.uniform2f(u.point, x, y); gl.uniform3f(u.value, vx, vy, 0);
      G.run(P.splat, vel.write); vel.swap();
    }
    u = G.use(P.curl); gl.uniform1i(u.uVelocity, G.bind(0, vel.read.tex)); G.run(P.curl, crl);
    u = G.use(P.vorticity);
    gl.uniform1i(u.uVelocity, G.bind(0, vel.read.tex)); gl.uniform1i(u.uCurl, G.bind(1, crl.tex));
    gl.uniform1f(u.curl, cfg.curl * calm.curl); gl.uniform1f(u.dt, dt);
    G.run(P.vorticity, vel.write); vel.swap();
    u = G.use(P.divergence); gl.uniform1i(u.uVelocity, G.bind(0, vel.read.tex)); G.run(P.divergence, div);
    u = G.use(P.scale); gl.uniform1i(u.uTexture, G.bind(0, prs.read.tex)); gl.uniform1f(u.value, .8);
    G.run(P.scale, prs.write); prs.swap();
    u = G.use(P.pressure); gl.uniform1i(u.uDivergence, G.bind(1, div.tex));
    for (let i = 0; i < cfg.pressureIters; i++) {
      gl.uniform1i(u.uPressure, G.bind(0, prs.read.tex));
      G.run(P.pressure, prs.write); prs.swap();
    }
    u = G.use(P.gradient);
    gl.uniform1i(u.uPressure, G.bind(0, prs.read.tex)); gl.uniform1i(u.uVelocity, G.bind(1, vel.read.tex));
    G.run(P.gradient, vel.write); vel.swap();
    u = G.use(P.advect);
    gl.uniform1i(u.uVelocity, G.bind(0, vel.read.tex)); gl.uniform1i(u.uSource, G.bind(0, vel.read.tex));
    gl.uniform1f(u.dt, dt); gl.uniform1f(u.dissipation, calm.dissipation);
    G.run(P.advect, vel.write); vel.swap();
    u = G.use(P.ink);
    gl.uniform1i(u.uVelocity, G.bind(0, vel.read.tex)); gl.uniform1i(u.uDisp, G.bind(1, ink.read.tex));
    gl.uniform2fv(u.velTexel, vel.texel); gl.uniform1f(u.dt, dt); gl.uniform1f(u.relax, Math.exp(-calm.relax * dt));
    G.run(P.ink, ink.write); ink.swap();
  }

  return { resize, splat, step, get ok() { return ok; }, get inkTex() { return ok ? ink.read.tex : null; } };
}
```

```js
import { FLUID, SCROLL } from "./config.js";
import { calmness, capFlick, rippleStrength } from "./logic.js";
import { createFluid } from "./fluid.js";

// Pointer and scroll input that moves water: stirring the name, stones, and the
// current that carries the name downstream as you scroll away.
export function createStirring(ctx) {
  const pick = v => (Array.isArray(v) ? v[ctx.coarse ? 1 : 0] : v);
  const cfg = Object.fromEntries(Object.entries(FLUID).map(([k, v]) => [k, pick(v)]));
  const fluid = createFluid(ctx.G, cfg);
  const { ptr } = ctx;
  let px = null, py = null, lx = 0, ly = 0, lastRip = 0, lastScroll = scrollY, active = true;

  addEventListener("pointermove", e => {
    const t = ctx.now();
    const wx = e.clientX, wy = e.clientY + scrollY;
    const d = Math.hypot(wx - lx, wy - ly);
    const speed = d / Math.max(t - ptr.lastMove, 1 / 120);
    if (px !== null && ctx.nameVis > .05) {
      const mx = e.clientX - px, my = e.clientY - py;
      if (Math.hypot(mx, my) < 400) {
        const [cx, cy] = capFlick(mx, my, cfg.flickCap);
        if (cx || cy) fluid.splat(e.clientX / ctx.W, 1 - e.clientY / ctx.H, cx / ctx.H * cfg.force, -cy / ctx.H * cfg.force);
      }
      ctx.emit("stir", speed);
    }
    px = e.clientX; py = e.clientY;
    ptr.x = wx; ptr.y = wy; ptr.lastMove = t;
    if (d > 26 && t - lastRip > .04) {
      ctx.ripples.add(wx, wy, rippleStrength(speed));
      lx = wx; ly = wy; lastRip = t;
    }
  });

  // a dropped stone: a ring of outward pushes plus a big ripple
  function stone(x, y) {
    if (ctx.nameVis > .05) {
      for (let i = 0; i < 10; i++) {
        const a = i / 10 * Math.PI * 2;
        fluid.splat((x + Math.cos(a) * 26) / ctx.W, 1 - (y + Math.sin(a) * 26) / ctx.H, Math.cos(a) * 560, -Math.sin(a) * 560, .0014);
      }
    }
    ctx.ripples.add(x, y + scrollY, 1.1);
    ctx.emit("stone");
  }

  addEventListener("pointerdown", e => {
    if (e.target.closest("a, button")) return;
    px = e.clientX; py = e.clientY;
    ptr.x = e.clientX; ptr.y = e.clientY + scrollY; ptr.lastMove = ctx.now();
    stone(e.clientX, e.clientY);
  });
  const lift = e => { if (e.pointerType !== "mouse") px = null; };
  addEventListener("pointerup", lift);
  addEventListener("pointercancel", lift);
  addEventListener("pointerout", e => { if (!e.relatedTarget) px = null; }); // left the window

  addEventListener("scroll", () => {
    const d = scrollY - lastScroll;
    lastScroll = scrollY;
    if (d > 0 && scrollY < ctx.H * SCROLL.pushUntil) {
      const r = ctx.els.name.getBoundingClientRect();
      const k = Math.min(d, 60) / ctx.H * cfg.force * 1.4;
      for (let i = 0; i < 6; i++) {
        const x = r.left + r.width * (i + .5) / 6, y = r.top + r.height * (.25 + Math.random() * .5);
        fluid.splat(x / ctx.W, 1 - y / ctx.H, k, -k * .15, .004);
      }
      ctx.lastScrollT = ctx.now();
    }
  }, { passive: true });

  return {
    splat: fluid.splat,
    resize() { fluid.resize(ctx.W, ctx.H); },
    update(t, dt) {
      active = scrollY < ctx.H * SCROLL.fluidUntil;
      if (active) fluid.step(dt, calmness(Math.min(t - ptr.lastMove, t - ctx.lastScrollT), cfg.velDissipation));
      ctx.F.inkTex = fluid.inkTex;
    },
    state() { return { fluidOK: fluid.ok, fluidActive: active }; },
  };
}
```

```js
// A quiet top-down silhouette in the river: wanders through the name, flees the pointer.
// World coordinates (page px). Skeleton: 8 spine nodes, 2 tail tips, 2 fins -> [x, y, r] x 12.
export function createFish({ coarse = false, random = Math.random } = {}) {
  const s = { x: -100, y: 0, a: 0, v: 120, phase: 0, target: null, retarget: 0, scaredUntil: 0, lastWake: 0 };
  const points = new Float32Array(36);

  function pickTarget(t, env) {
    const r = env.nameRect;
    const side = random() < .5 ? -1 : 1;
    s.target = {
      x: env.W / 2 + side * (env.W * (.25 + random() * .3)),
      y: r.y + r.h * (.15 + random() * .7) + (random() < .25 ? (random() - .5) * env.H * .6 : 0),
    };
    s.retarget = t + 4 + random() * 4;
  }

  function step(t, dt, env) {
    const { W, H, ptr } = env;
    const L = Math.max(42, Math.min(96, W * .052));
    const m = 80;
    const dxm = s.x - ptr.x, dym = s.y - ptr.y;
    const dm = Math.hypot(dxm, dym) || 1;
    if (dm < 150 + L && t - ptr.lastMove < .25 && t > s.scaredUntil - .5) {
      if (t > s.scaredUntil) env.ripple(s.x, s.y, .55);
      s.scaredUntil = t + .9;
      s.target = { x: s.x + dxm / dm * 500, y: s.y + dym / dm * 500 };
      s.retarget = t + 1.2;
    }
    const scared = t < s.scaredUntil;
    if (!s.target || t > s.retarget || Math.hypot(s.target.x - s.x, s.target.y - s.y) < 60) pickTarget(t, env);
    let want = Math.atan2(s.target.y - s.y, s.target.x - s.x);
    if (!scared && (s.x < -m || s.x > W + m || s.y < -m || s.y > H + m)) want = Math.atan2(H / 2 - s.y, W / 2 - s.x);
    s.a = angleTo(s.a, want, (scared ? 7 : 1.6) * dt);
    const vt = (scared ? 620 : 115) * (coarse ? .8 : 1);
    s.v += (vt - s.v) * Math.min(1, dt * (scared ? 6 : 1.2));
    s.x += Math.cos(s.a) * s.v * dt;
    s.y += Math.sin(s.a) * s.v * dt;
    s.phase += dt * (5 + s.v * .028);
    if (s.v > 260 && t - s.lastWake > .12) { env.ripple(s.x, s.y, .12); s.lastWake = t; }
    else if (t - s.lastWake > 1.4) { env.ripple(s.x - Math.cos(s.a) * L * .8, s.y - Math.sin(s.a) * L * .8, .05); s.lastWake = t; }
    env.drag(s.x, s.y, Math.cos(s.a), Math.sin(s.a), s.v * dt);

    const ca = Math.cos(s.a), sa = Math.sin(s.a);
    const amp = L * .07 * Math.min(1.6, .6 + s.v / 300);
    const toWorld = (lx, ly) => [s.x + lx * ca - ly * sa, s.y + lx * sa + ly * ca];
    const local = [];
    for (let i = 0; i < 8; i++) {
      const u = i / 7;
      const lx = -u * L * .82;
      const ly = Math.sin(s.phase - u * 3.2) * amp * u * u;
      const r = L * (.03 + .1 * Math.pow(Math.sin(Math.min(1, u * 1.25 + .06) * Math.PI), .75));
      local.push([lx, ly]);
      points.set([...toWorld(lx, ly), Math.max(r, L * .018)], i * 3);
    }
    const [tx, ty] = local[7], [px, py] = local[6];
    const ta = Math.atan2(ty - py, tx - px);
    for (const [k, side] of [[8, 1], [9, -1]]) {
      const aa = ta + side * .7;
      points.set([...toWorld(tx + Math.cos(aa) * L * .24, ty + Math.sin(aa) * L * .24), L * .01], k * 3);
    }
    const [fx, fy] = local[2];
    const flap = Math.sin(s.phase * .5) * .2;
    for (const [k, side] of [[10, 1], [11, -1]]) {
      const aa = Math.PI - side * (1.15 + flap);
      points.set([...toWorld(fx + Math.cos(aa) * L * .17, fy + Math.sin(aa) * L * .17), L * .012], k * 3);
    }
  }

  return { state: s, points, step };
}

// Wires the fish into the page: hidden when motion is reduced; drags ink when on screen.
export function fishFeature(ctx, splat) {
  const fish = createFish({ coarse: ctx.coarse });
  fish.state.y = ctx.H * .4;
  const env = {
    W: 0, H: 0, nameRect: null, ptr: ctx.ptr,
    ripple: (x, y, s) => ctx.ripples.add(x, y, s),
    drag(x, y, dx, dy, dist) {
      const sy = y - scrollY;
      if (x > 0 && x < env.W && sy > 0 && sy < env.H) {
        const k = dist * FLUID.force / env.H * .35;
        splat(x / env.W, 1 - sy / env.H, dx * k, -dy * k, .0012);
      }
    },
  };
  return {
    update(t, dt) {
      ctx.F.fishOn = ctx.still ? 0 : 1;
      if (ctx.still) return;
      env.W = ctx.W; env.H = ctx.H; env.nameRect = ctx.world.nameRect;
      fish.step(t, dt, env);
      ctx.F.fishPts = fish.points;
    },
    state() { return { fishOn: ctx.F.fishOn === 1, fish: { x: fish.state.x, y: fish.state.y } }; },
  };
}
```

```js
// Hovering or focusing a word deepens its carving; the bottle lifts out of the sand.
export function beachFeature(ctx) {
  let hoverIdx = -1, bottleHover = false, shown = [0, 0, 0, 0];

  ctx.els.words.forEach((el, i) => {
    const on = () => { hoverIdx = i; };
    const off = () => { if (hoverIdx === i) hoverIdx = -1; };
    for (const [ev, fn] of [["pointerenter", on], ["pointerleave", off], ["focus", on], ["blur", off]]) el.addEventListener(ev, fn);
  });
  for (const [ev, v] of [["pointerenter", true], ["pointerleave", false], ["focus", true], ["blur", false]]) {
    ctx.els.bottle.addEventListener(ev, () => { bottleHover = v; });
  }

  return {
    update(t, dt) {
      const F = ctx.F;
      F.hoverAmt += ((hoverIdx >= 0 ? 1 : 0) - F.hoverAmt) * Math.min(1, dt * 8);
      if (hoverIdx >= 0) shown = ctx.world.rects[hoverIdx] || shown;
      F.hoverRect = shown;
      F.bottleLift += ((bottleHover ? 1 : 0) - F.bottleLift) * Math.min(1, dt * 7);
    },
    state() { return { hoverIdx, hoverAmt: ctx.F.hoverAmt, bottleLift: ctx.F.bottleLift }; },
  };
}
```

```python
"""Synthesize the site's sounds with ffmpeg + plain Python (no samples, so no licence questions).

Writes audio/{river,surf,swish,plop}.{webm,mp3}. Loops are made seamless by
crossfading their tail into their head.
"""
import array
import math
import random
import subprocess
import tempfile
import wave
from pathlib import Path

SR = 48_000
OUT = Path(__file__).resolve().parents[1] / "audio"


def noise(seconds, color, filters, seed):
    """Filtered noise from ffmpeg as a list of floats in [-1, 1]."""
    cmd = ["ffmpeg", "-loglevel", "error", "-f", "lavfi", "-i", f"anoisesrc=d={seconds}:c={color}:r={SR}:a=0.5:seed={seed}",
           "-af", filters, "-ac", "1", "-ar", str(SR), "-f", "s16le", "-"]
    raw = subprocess.run(cmd, capture_output=True, check=True).stdout
    return [v / 32768 for v in array.array("h", raw)]


def seamless(x, fade_s):
    """Equal-power crossfade of the last fade_s seconds into the first, returning len(x) - fade samples."""
    f = int(fade_s * SR)
    body = x[:len(x) - f]
    for i in range(f):
        w = i / f * math.pi / 2
        body[i] = body[i] * math.sin(w) + x[len(x) - f + i] * math.cos(w)
    return body


def normalize(x, peak):
    m = max(abs(v) for v in x) or 1
    return [v / m * peak for v in x]


def write(name, x, bitrate_opus="48k", bitrate_mp3="64k"):
    with tempfile.TemporaryDirectory() as tmp:
        wav = Path(tmp) / f"{name}.wav"
        with wave.open(str(wav), "wb") as w:
            w.setnchannels(1)
            w.setsampwidth(2)
            w.setframerate(SR)
            w.writeframes(array.array("h", (int(max(-1, min(1, v)) * 32767) for v in x)).tobytes())
        for ext, codec, rate in (("webm", "libopus", bitrate_opus), ("mp3", "libmp3lame", bitrate_mp3)):
            subprocess.run(["ffmpeg", "-loglevel", "error", "-y", "-i", str(wav), "-c:a", codec, "-b:a", rate,
                            str(OUT / f"{name}.{ext}")], check=True)


def river():
    x = noise(23, "brown", "highpass=f=70,lowpass=f=900,tremolo=f=0.25:d=0.25", seed=7)
    return normalize(seamless(x, 3), .7)


def surf():
    x = noise(27, "pink", "highpass=f=120,lowpass=f=1400", seed=11)
    # three waves per 24 s loop (period 8 s divides the loop, so the crossfade lines up)
    env = [.18 + .82 * max(0.0, math.sin(2 * math.pi * (i / SR) / 8)) ** 3 for i in range(len(x))]
    return normalize(seamless([a * b for a, b in zip(x, env)], 3), .7)


def swish():
    x = noise(.7, "pink", "bandpass=f=1100:width_type=h:w=900", seed=3)
    n = len(x)
    env = [min(1, (i / SR) / .08) * math.exp(-max(0, i / SR - .08) * 6) * min(1, (n - i) / (SR * .05)) for i in range(n)]
    return normalize([a * b for a, b in zip(x, env)], .8)


def plop():
    rng = random.Random(5)
    out, phase = [], 0.0
    for i in range(int(.35 * SR)):
        t = i / SR
        f = 520 + 700 * (1 - math.exp(-t * 40))  # a bubble's pitch rises quickly as it closes
        phase += 2 * math.pi * f / SR
        tone = math.sin(phase) * math.exp(-t * 14)
        click = rng.uniform(-1, 1) * math.exp(-t * 300) * .5
        out.append(tone + click)
    return normalize(out, .85)


if __name__ == "__main__":
    OUT.mkdir(exist_ok=True)
    for name, make in (("river", river), ("surf", surf), ("swish", swish), ("plop", plop)):
        write(name, make())
```

- [ ] **Step 4: Write `js/audio.js`**

```js
import { equalPowerGains, swishGain, readMuted, writeMuted } from "./logic.js";

const safeStorage = () => { try { return window.localStorage; } catch { return null; } };

// River murmur at the top crossfades into surf at the beach; swishes and plops on top.
// Nothing is created until the first user gesture (browsers block audio before that).
export function createAudio({ cfg, base = "audio/", storage = safeStorage() }) {
  let ctx = null, master, riverGain, surfGain, buffers = null, lastSwish = -1, progress = 0;
  let muted = readMuted(storage, cfg.storageKey);
  const ext = new Audio().canPlayType('audio/webm; codecs="opus"') ? "webm" : "mp3";

  const load = async name => {
    const res = await fetch(`${base}${name}.${ext}`);
    if (!res.ok) throw new Error(`${name}: HTTP ${res.status}`);
    return ctx.decodeAudioData(await res.arrayBuffer());
  };

  function fadeTo(v, secs) {
    const t = ctx.currentTime;
    master.gain.cancelScheduledValues(t);
    master.gain.setValueAtTime(master.gain.value, t);
    master.gain.linearRampToValueAtTime(v, t + secs);
  }

  function loop(buffer, dest) {
    const s = ctx.createBufferSource();
    s.buffer = buffer;
    s.loop = true;
    s.connect(dest);
    s.start();
  }

  function applyProgress() {
    if (!ctx) return;
    const [r, s] = equalPowerGains(progress);
    riverGain.gain.setTargetAtTime(r, ctx.currentTime, .15);
    surfGain.gain.setTargetAtTime(s, ctx.currentTime, .15);
  }

  function unlock() {
    if (ctx) return;
    const AC = window.AudioContext || window.webkitAudioContext;
    if (!AC) return;
    ctx = new AC();
    master = ctx.createGain();
    master.gain.value = 0;
    master.connect(ctx.destination);
    riverGain = ctx.createGain();
    surfGain = ctx.createGain();
    riverGain.connect(master);
    surfGain.connect(master);
    applyProgress();
    Promise.all(cfg.names.map(load))
      .then(([river, surf, swish, plop]) => {
        buffers = { river, surf, swish, plop };
        loop(river, riverGain);
        loop(surf, surfGain);
        fadeTo(muted ? 0 : 1, cfg.fadeIn);
      })
      .catch(err => console.warn("audio unavailable:", err));
  }

  function oneShot(buffer, gain, rate) {
    if (!ctx || !buffer || muted) return;
    const s = ctx.createBufferSource();
    s.buffer = buffer;
    s.playbackRate.value = rate;
    const g = ctx.createGain();
    g.gain.value = gain;
    s.connect(g).connect(master);
    s.start();
  }

  return {
    unlock,
    setProgress(p) { progress = p; applyProgress(); },
    swish(speed, nowSeconds) {
      const g = swishGain(speed);
      if (g < .02 || nowSeconds - lastSwish < cfg.swishGap) return;
      lastSwish = nowSeconds;
      oneShot(buffers?.swish, g, .9 + Math.random() * .2);
    },
    plop() { oneShot(buffers?.plop, .8, 2 ** ((Math.random() * 400 - 200) / 1200)); },
    setMuted(v) {
      muted = v;
      writeMuted(storage, cfg.storageKey, v);
      if (buffers) fadeTo(v ? 0 : 1, .4);
    },
    suspend() { ctx?.suspend(); },
    resume() { ctx?.resume(); },
    get muted() { return muted; },
    get started() { return !!ctx; },
    get state() { return ctx ? ctx.state : "none"; },
    get ready() { return !!buffers; },
  };
}
```

```js
import { AUDIO } from "./config.js";
import { scrollProgress } from "./logic.js";
import { createAudio } from "./audio.js";

// The mute button, the first-gesture unlock, scroll crossfade and hidden-tab pause.
export function setupSound(button, on) {
  const audio = createAudio({ cfg: AUDIO });
  const progress = () => scrollProgress(scrollY, document.documentElement.scrollHeight - innerHeight);
  const sync = () => button.setAttribute("aria-pressed", String(audio.muted));
  sync();
  button.hidden = false;

  const unlock = () => {
    audio.unlock();
    audio.setProgress(progress());
    removeEventListener("pointerdown", unlock, true);
    removeEventListener("keydown", unlock, true);
  };
  addEventListener("pointerdown", unlock, true);
  addEventListener("keydown", unlock, true);

  button.addEventListener("click", () => { audio.setMuted(!audio.muted); sync(); });
  addEventListener("scroll", () => audio.setProgress(progress()), { passive: true });
  document.addEventListener("visibilitychange", () => (document.hidden ? audio.suspend() : audio.resume()));
  on("stir", speed => audio.swish(speed, performance.now() / 1000));
  on("stone", () => audio.plop());
  return audio;
}
```

```python
"""Measure real frame rate (normal mode, real clock) for 5 seconds.

Usage: .venv/bin/python tools/perf.py [--headed] [--size 1440x900] [--phone]
Headless runs use software WebGL and only show relative cost; use --headed for real GPU numbers.
"""
import argparse
import functools
import http.server
import statistics
import sys
import threading
from pathlib import Path

from playwright.sync_api import sync_playwright

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "tests" / "e2e"))
from conftest import CHROMIUM_ARGS, QuietHandler  # noqa: E402

FRAMES = """() => new Promise(done => {
  const ts = [];
  const f = t => { ts.push(t); ts.length < 300 ? requestAnimationFrame(f) : done(ts); };
  requestAnimationFrame(f);
})"""


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--headed", action="store_true")
    ap.add_argument("--size", default="1440x900")
    ap.add_argument("--phone", action="store_true")
    a = ap.parse_args()
    w, h = (390, 844) if a.phone else map(int, a.size.split("x"))
    httpd = http.server.ThreadingHTTPServer(("127.0.0.1", 0), functools.partial(QuietHandler, directory=str(ROOT)))
    threading.Thread(target=httpd.serve_forever, daemon=True).start()
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=not a.headed, args=[] if a.headed else CHROMIUM_ARGS)
        page = browser.new_page(viewport={"width": w, "height": h}, is_mobile=a.phone, has_touch=a.phone)
        page.goto(f"http://127.0.0.1:{httpd.server_port}/index.html")
        page.wait_for_timeout(1500)
        for label, y in (("hero", 0), ("beach", 10 ** 6)):
            page.evaluate(f"scrollTo(0, {y})")
            page.wait_for_timeout(300)
            ts = page.evaluate(FRAMES)
            gaps = sorted(b - a for a, b in zip(ts, ts[1:]))
            print(f"{label:6} median {1000 / statistics.median(gaps):5.1f} fps   p95 frame {gaps[int(len(gaps) * .95)]:5.1f} ms")
        browser.close()
    httpd.shutdown()


if __name__ == "__main__":
    main()
```
